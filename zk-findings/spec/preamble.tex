% Source citations are long monospace paths. \path is fragile inside other
% macros' arguments, so instead walk the token list (expl3) and insert a
% discretionary break after every / . , - so a citation can wrap inside a
% table cell or a citation-dense paragraph. \_ (escaped underscore) is a single
% control-sequence token and is left untouched.
\ExplSyntaxOn
\NewDocumentCommand{\src}{m}{
  \tl_set:Nn \l_tmpa_tl {#1}
  \tl_replace_all:Nnn \l_tmpa_tl {/} {/\allowbreak}
  \tl_replace_all:Nnn \l_tmpa_tl {.} {.\allowbreak}
  \tl_replace_all:Nnn \l_tmpa_tl {,} {,\allowbreak}
  \tl_replace_all:Nnn \l_tmpa_tl {-} {-\allowbreak}
  \allowbreak{\small\textcolor{gray}{[\texttt{\tl_use:N \l_tmpa_tl}]}}\allowbreak
}
\ExplSyntaxOff
